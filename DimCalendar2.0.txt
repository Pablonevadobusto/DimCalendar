if [ID] = 1 then Table1 else
if [ID] = 2 then Table2 else
if [ID] = 3 then Table3 else
if [ID] = 4 then Table4 else
null),
   Custom2 = Table.SelectRows(#"Added Custom", each
Type.Is(Value.Type([Custom]), type table) = true),

    #"Expanded Schema" = Table.ExpandTableColumn(#"Added Custom1", "Schema", {"Name", "Kind"}, {"Name", "Kind"}),

 // Creating the column Year as well as changing the default data type from number to Int64.Type
